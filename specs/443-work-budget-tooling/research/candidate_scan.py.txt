def resolve_rel(p,level,module):
  pk=modname[p].split('.')
  if os.path.basename(p)!='__init__.py': pk=pk[:-1]
  if level: pk=pk[:len(pk)-(level-1)]
  base='.'.join(pk)
  return base+('.'+module if module else '') if level else module

      for a in n.names:
        tab[a.asname or a.name]=(mod,a.name)
    elif isinstance(n,ast.Import):
      for a in n.names: tab[(a.asname or a.name).split('.')[0]]=(a.name,None)
  imps[p]=tab
def lookup_in_module(mod,name,seen=None):
  seen=seen or set()
  f=mod2file.get(mod)
  if not f or (mod,name) in seen: return []
  seen.add((mod,name))
  if name in toplevel[f]: return [toplevel[f][name]]
  # re-export
  tab=imps[f]
  if name in tab:
    m2,n2=tab[name]
    if n2: 
      r=lookup_in_module(m2,n2,seen)
      if r: return r
  return []
NOISY={'__init__','add','get','append','extend','pop','items','keys','values','copy','update','at','find','count','line','ok','rect','build','index','sort','join','format'}
def resolve(p,qual,call):
  out=[]
  f=call.func
  if isinstance(f,ast.Name):
    nm=f.id
    # nested def in same func?
    for (pp,qq) in [(p,qual+'.'+nm)]:
      if (pp,qq) in defs: out.append((pp,qq))
    if nm in toplevel[p]: out.append(toplevel[p][nm])
    elif nm in imps[p]:
      m,n=imps[p][nm]
      if n: out+=lookup_in_module(m,n)
  elif isinstance(f,ast.Attribute):
    nm=f.attr
    if isinstance(f.value,ast.Name) and f.value.id in imps[p]:
      m,n=imps[p][f.value.id]
      full=m+('.'+n if n else '')
      r=lookup_in_module(full,nm)
      if r: return r
      if mod2file.get(full) : return []
    if nm in NOISY: return []
    if isinstance(f.value,ast.Name) and f.value.id=='self':
      cls=qual.rsplit('.',1)[0] if '.' in qual else ''
      if (p,cls+'.'+nm) in defs: return [(p,cls+'.'+nm)]
    out+=[m for m in methods.get(nm,[])]
  return out

def edges(k):
  p,q=k; n=defs[k]; res=set()
  for x in ast.walk(n):
    if isinstance(x,ast.Call): res.update(resolve(p,q,x))
    elif isinstance(x,ast.Name) and isinstance(x.ctx,ast.Load):
      if x.id in toplevel[p] and toplevel[p][x.id] in defs: res.add(toplevel[p][x.id])
      elif x.id in imps[p] and imps[p][x.id][1]:
        res.update(r for r in lookup_in_module(*imps[p][x.id]) if r in defs)
  res.discard(k)
  return res

def cname(c):

def own_tests(node):
  s=set()
  for x in ast.walk(node):
    if isinstance(x,ast.Call):
      n=cname(x)
      if n and isname(n): s.add(n)
  return s

def cands(k):
  p,q=k; n=defs[k]; out=[]
  for x in ast.walk(n):
    kind=None
    if isinstance(x,LOOPS): kind=type(x).__name__
    elif isinstance(x,ast.Call) and cname(x) in ('min','max','sorted','next') and isinstance(x.func,ast.Name): kind=cname(x)
    if not kind: continue
    # skip loops belonging to nested defs (they're keyed separately)
    d=own_tests(x)
    # drop self-evident
    if d: out.append((x.lineno,kind,sorted(d),'direct'))
    else:
      hs=set()
      for y in ast.walk(x):
        if isinstance(y,ast.Call):
          nm=cname(y)
          if nm in HELP and nm not in NOISY: hs.add(nm)
      if hs: out.append((x.lineno,kind,sorted(hs),'via'))
  return out

def resolve_strong(p,qual,call):
  f=call.func
  if isinstance(f,ast.Attribute):
    nm=f.attr
    if isinstance(f.value,ast.Name):
      if f.value.id in imps[p]:
        m,n=imps[p][f.value.id]; full=m+('.'+n if n else '')
        return lookup_in_module(full,nm)
      if f.value.id=='self':
        cls=qual.rsplit('.',1)[0] if '.' in qual else ''
        if (p,cls+'.'+nm) in defs: return [(p,cls+'.'+nm)]
        return [] 
    # s.method / settlement method: resolve only if unique method of that name
    if nm not in NOISY and len(methods.get(nm,[]))==1: return list(methods[nm])
    return []
  return resolve(p,qual,call)
